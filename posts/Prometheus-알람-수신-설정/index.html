# Prometheus 알람 수신 설정

  - 흐름 : Prometheus -> AlertManager -> Slack

  - 알람 발송 할 Slack webhook URL 추가 value 파일 적용

prometheus:

&nbsp; service:

&nbsp;   type: NodePort          # Prometheus UI를 NodePort로 외부 접속 가능하게 설정

&nbsp;                                 # (minikube IP:NodePort 로 접근 가능)



grafana:

&nbsp; service:

&nbsp;   type: NodePort          # Grafana UI를 NodePort로 외부 접속 가능하게 설정

&nbsp; env:

&nbsp;   GF\_SECURITY\_ADMIN\_USER: "admin"       # Grafana 관리자 계정 아이디

&nbsp;   GF\_SECURITY\_ADMIN\_PASSWORD: "admin"   # Grafana 관리자 계정 비밀번호



alertmanager:

&nbsp; service:

&nbsp;   type: NodePort          # Alertmanager UI를 NodePort로 외부 접속 가능하게 설정



&nbsp; alertmanagerSpec:

&nbsp;   useExistingSecret: false  # 기존 Kubernetes Secret을 사용하지 않고  Helm values에 정의된 config를 기반으로 Secret을 자동 생성함

&nbsp;   

&nbsp; config:

&nbsp;   global:

&nbsp;     resolve\_timeout: 5m   # Alert가 resolve(해소)된 후 상태를 유지하는 기본 타임아웃 , 5분 동안 추가 변경 없으면 resolve 처리



&nbsp;   route:

&nbsp;     receiver: slack-notifications  # 기본 수신자 설정 (매칭되는 route 없을 경우 이 receiver로 전달)



&nbsp;     group\_by:

&nbsp;       - alertname  # 동일 alertname 기준으로 알람 묶음 (Slack 메시지 그룹핑 기준)

&nbsp;     group\_wait: 30s    # 알람 발생 후 처음 전송 전까지 대기 시간 (여러 알람을 묶기 위해 30초 대기)

&nbsp;     group\_interval: 5m # 같은 그룹에서 새 알람이 생겼을 때 다음 알람을 보내기 전 최소 대기 시간

&nbsp;     repeat\_interval: 4h  # 동일한 FIRING 알람을 재전송하는 주기 (4시간마다 한 번씩 재알림)

&nbsp;     routes: \[] # 하위 라우팅 규칙 정의 영역 현재는 비어있음 (severity별 분기 등 가능)



&nbsp;   receivers:

&nbsp;     - name: slack-notifications # route에서 지정한 receiver 이름과 동일해야 함

&nbsp;       slack\_configs:

&nbsp;         - api\_url: "Slack 알람 수신 웹훅 URL 주소"  # Slack Incoming Webhook URL

&nbsp;           channel: "#alert\_test\_msp" # Slack 알림이 전송될 채널 이름

&nbsp;           send\_resolved: true  # Alert가 해소(RESOLVED) 되었을 때도 Slack 알림 전송 여부

## 2. helm 반영

  - helm 릴리즈 이름 확인


helm list -A

  - helm 업그레이드로 설정 적용


helm upgrade wb-prometheus prometheus-community/kube-prometheus-stack -f prometheus-value.yaml -n monitoring

apiVersion: monitoring.coreos.com/v1

kind: PrometheusRule

metadata:

&nbsp; name: cpu-test-alert

&nbsp; namespace: monitoring

&nbsp; labels:

&nbsp;   release: wb-prometheus

spec:

&nbsp; groups:

&nbsp;   - name: test-alert

&nbsp;     rules:

&nbsp;       - alert: CPUHighTest

&nbsp;         expr: vector(1)

&nbsp;         for: 30s

&nbsp;         labels:

&nbsp;           severity: critical

&nbsp;         annotations:

&nbsp;           summary: "CPU Test Alert"

&nbsp;           description: "This is a manual Slack test alert."
